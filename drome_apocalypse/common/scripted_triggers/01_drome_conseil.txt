# Assemblee generale : au moins 350 communes (environ la moitie) ont vote oui
conseil_adopte = {
	calc_true_if = {
		all_country = {
			has_country_flag = conseil_oui
		}
		amount = 500
	}
}

# Conseil : majorite de 6 sieges sur 10
conseil_siege_majorite = {
	calc_true_if = {
		all_country = {
			has_country_flag = conseil_siege_oui
		}
		amount = 6
	}
}

# Veto : un membre permanent a vote non
conseil_veto = {
	any_country = {
		has_country_flag = ccl_permanent
		NOT = { has_country_flag = conseil_siege_oui }
	}
}
