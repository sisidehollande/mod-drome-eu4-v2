# Debarquements : la horde prend pied sur une commune cotiere
country_event = {
	id = drome_horde.10
	title = drome_horde.10.t
	desc = drome_horde.10.d
	picture = SIEGE_eventPicture
	hidden = yes
	trigger = {
		tag = ZOM
	}
	mean_time_to_happen = {
		days = 60
		modifier = {
			factor = 2
			has_global_flag = horde_facile
		}
		modifier = {
			factor = 0.5
			has_global_flag = horde_brutale
		}
	}
	immediate = {
		random_province = {
			limit = {
				is_city = yes
				any_neighbor_province = {
					OR = {
						province_id = 722
						province_id = 710
						province_id = 711
						province_id = 723
					}
				}
				NOT = { owned_by = ZOM }
			}
			horde_convertir_province = yes
		}
	}
	option = {
		name = drome_horde.10.a
	}
}

# Les communes voisines declarent la guerre a la horde
country_event = {
	id = drome_horde.12
	title = drome_horde.12.t
	desc = drome_horde.12.d
	picture = BATTLE_eventPicture
	hidden = yes
	trigger = {
		tag = ZOM
	}
	mean_time_to_happen = {
		days = 12
	}
	immediate = {
		random_neighbor_country = {
			limit = {
				NOT = { tag = ZOM }
				NOT = { war_with = ZOM }
			}
			declare_war_with_cb = {
				who = ZOM
				casus_belli = cb_horde
			}
		}
	}
	option = {
		name = drome_horde.12.a
	}
}

# Victoire : la horde est eteinte
country_event = {
	id = drome_horde.20
	title = drome_horde.20.t

			}
		}
	}
	option = {
		name = drome_horde.20.a
		add_prestige = 50
	}
}
